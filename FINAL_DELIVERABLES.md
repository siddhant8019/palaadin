### Start Development:
```bash
# Validate environment
cd backend && node scripts/validate-env.js

# Start backend
npm run dev

# Start frontend (new terminal)
cd frontend && npm run dev

# Access at:
# Frontend: http://localhost:3000
# Backend:  http://localhost:5000 (or 4000)
# API Docs: http://localhost:5000/api-docs
```

### Run Tests:
```bash
# Backend
cd backend
npm test                 # Unit tests
npm run test:coverage    # With coverage
npm run test:integration # Integration

# Frontend
cd frontend
npm test                 # Unit tests
npm run test:e2e         # E2E with Playwright
```

### Use New Features:
```typescript
// Caching
import { cacheService, CacheKeys } from '@/services/cache.service';
const data = await cacheService.getOrSet(
  CacheKeys.companyList('tech'),
  () => fetchCompanies(),
  3600
);

// Background Jobs
import { queueService } from '@/services/queue.service';
await queueService.addScrapingJob({
  url: 'https://example.com',
  userId: user.id
});

// Transactions
import { TransactionManager } from '@/utils/transaction';
await TransactionManager.execute(async (manager) => {
  await manager.save(person);
  await manager.save(company);
});

// Error Boundaries
import { ErrorBoundary } from '@/components/error-boundary/ErrorBoundary';
<ErrorBoundary>
  <YourComponent />
</ErrorBoundary>
```
